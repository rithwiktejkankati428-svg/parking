/**
 * Handles database connectivity using Singleton pattern.
 */
public class DatabaseConnection {
    private static final String URL = "jdbc:mysql://localhost:3306/smartpark_db";
    private static final String USER = "root"; // Update for local setup
    private static final String PASSWORD = "1234"; // Update for local setup

    private static Connection connection = null;

    private DatabaseConnection() {
        // private constructor for Singleton
    }

    public static Connection getConnection() {
        try {
            if (connection == null || connection.isClosed()) {
                // Register JDBC driver (optional in newer JDBC but good for APP project)
                Class.forName("com.mysql.cj.jdbc.Driver");
                connection = DriverManager.getConnection(URL, USER, PASSWORD);
            }
        } catch (ClassNotFoundException e) {
            System.err.println("MySQL JDBC Driver not found. Add connector jar to classpath.");
            e.printStackTrace();
        } catch (SQLException e) {
            System.err.println("Database connection failed. Ensure MySQL is running and credentials are correct.");
            e.printStackTrace();
        }
        return connection;
    }

    public static void closeConnection() {
        try {
            if (connection != null && !connection.isClosed()) {
                connection.close();
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}

public class MigrationService {

    public static void runMigrations() {
        System.out.println("[MIGRATION] Checking database schema...");
        
        String[] alterStatements = {
            "ALTER TABLE users ADD COLUMN google_subject_id VARCHAR(255) UNIQUE",
            "ALTER TABLE users ADD COLUMN dob DATE",
            "ALTER TABLE users ADD COLUMN gender VARCHAR(10)",
            "ALTER TABLE users ADD COLUMN mobile VARCHAR(20)",
            "ALTER TABLE users ADD COLUMN personal_email VARCHAR(100)",
            "ALTER TABLE users ADD COLUMN address TEXT",
            "ALTER TABLE users ADD COLUMN department VARCHAR(100)",
            "ALTER TABLE users ADD COLUMN program VARCHAR(100)",
            "ALTER TABLE users ADD COLUMN year INT",
            "ALTER TABLE users ADD COLUMN section VARCHAR(10)",
            "ALTER TABLE users ADD CONSTRAINT uc_google_email UNIQUE (google_email)",
            "ALTER TABLE users ADD CONSTRAINT uc_college_id UNIQUE (college_id)",
            "DELETE FROM users WHERE college_id LIKE 'STU-%' AND department = 'General'"
        };

        try (Connection conn = DatabaseConnection.getConnection()) {
            if (conn == null) return;
            
            try (Statement stmt = conn.createStatement()) {
                for (String sql : alterStatements) {
                    try {
                        stmt.execute(sql);
                        System.out.println("[MIGRATION] Successfully executed: " + sql);
                    } catch (SQLException e) {
                        // 1060 is "Duplicate column name". 1061 is "Duplicate key name". Safe to ignore.
                        if (e.getErrorCode() != 1060 && e.getErrorCode() != 1061) {
                            System.err.println("[MIGRATION ERROR] Failed to execute: " + sql);
                            e.printStackTrace();
                        }
                    }
                }
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
        System.out.println("[MIGRATION] Schema check complete.");
    }
}

package dao;

import interfaces.DatabaseOperations;
import model.User;
import model.Student;
import model.Faculty;
import model.Visitor;
import model.SecurityStaff;
import model.Admin;
import database.DatabaseConnection;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

public class UserDAO implements DatabaseOperations<User> {

    // Simple SHA-256 validation (for project demo purposes, we do direct matching as DB holds hashes)
    // Note: A real system uses BCrypt, but we're keeping it simple for B.Tech project
    public User login(String username, String passwordHash) {
        String sql = "SELECT * FROM users WHERE username = ? AND password_hash = ? AND is_active = TRUE";
        try (Connection conn = DatabaseConnection.getConnection()) {
            if (conn == null) return null;
            try (PreparedStatement pstmt = conn.prepareStatement(sql)) {
                pstmt.setString(1, username);
                pstmt.setString(2, passwordHash);
                ResultSet rs = pstmt.executeQuery();
                
                if (rs.next()) {
                    return mapRowToUser(rs);
                }
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return null;
    }

    private User mapRowToUser(ResultSet rs) throws SQLException {
        int id = rs.getInt("id");
        String name = rs.getString("name");
        String username = rs.getString("username");
        String role = rs.getString("role");
        String collegeId = rs.getString("college_id");

        switch (role) {
            case "STUDENT": 
                Student s = new Student(id, name, username, collegeId);
                try {
                    s.setGoogleSubjectId(rs.getString("google_subject_id"));
                    s.setGoogleEmail(rs.getString("google_email"));
                    s.setDob(rs.getDate("dob"));
                    s.setGender(rs.getString("gender"));
                    s.setMobile(rs.getString("mobile"));
                    s.setPersonalEmail(rs.getString("personal_email"));
                    s.setAddress(rs.getString("address"));
                    s.setDepartment(rs.getString("department"));
                    s.setProgram(rs.getString("program"));
                    s.setYear(rs.getInt("year"));
                    s.setSection(rs.getString("section"));
                } catch (SQLException ignore) {
                    // Ignore if columns are not present in some queries
                }
                return s;
            case "FACULTY": return new Faculty(id, name, username, collegeId);
            case "VISITOR": return new Visitor(id, name, username, collegeId);
            case "SECURITY": return new SecurityStaff(id, name, username, collegeId);

     */
    public ParkingSlot getRecommendedSlot(User user, Building destination, String vehicleType) {

        if (availableSlots.isEmpty()) {
            return null;
        }

        boolean isEv = vehicleType.equals("EV");

        for (ParkingSlot slot : availableSlots) {
            double score = 1000.0;
            
            // Calculate Euclidean distance (using simple coordinates)
            double distance = Math.sqrt(Math.pow(slot.getLocationX() - destination.getLocationX(), 2) + 
                                        Math.pow(slot.getLocationY() - destination.getLocationY(), 2));
            
            score -= (distance * 10); // Penalty for distance

            if (isEv && slot.isEv()) {
                score += 500; // Big bonus for matching EV to EV slot
            }

            if (user.getRole().equals("FACULTY")) {
                score += 100; // Small bonus priority for Faculty
            }

            slot.setRecommendationScore(score);
        }

        // Sort descending by score
        availableSlots.sort((s1, s2) -> Double.compare(s2.getRecommendationScore(), s1.getRecommendationScore()));

        return availableSlots.get(0); // Return highest score slot
    }

    private BufferedImage invertImage(BufferedImage src) {
        BufferedImage dest = new BufferedImage(src.getWidth(), src.getHeight(), src.getType());
        for (int x = 0; x < src.getWidth(); x++) {
            for (int y = 0; y < src.getHeight(); y++) {
                int rgba = src.getRGB(x, y);
                int a = (rgba >> 24) & 0xff;
                int r = 255 - ((rgba >> 16) & 0xff);
                int g = 255 - ((rgba >> 8) & 0xff);
                int b = 255 - (rgba & 0xff);
                dest.setRGB(x, y, (a << 24) | (r << 16) | (g << 8) | b);
            }
        }
        return dest;
    }

    private BufferedImage upscale(BufferedImage src, double scale) {

        BufferedImage dest = new BufferedImage(w, h, type);

    private BufferedImage rotateImage(BufferedImage src, double degrees) {
        double radians = Math.toRadians(degrees);
        double sin = Math.abs(Math.sin(radians));
        double cos = Math.abs(Math.cos(radians));
        int w = src.getWidth();
        int h = src.getHeight();
        int newW = (int) Math.floor(w * cos + h * sin);
        int newH = (int) Math.floor(h * cos + w * sin);
        int type = src.getType() == 0 ? BufferedImage.TYPE_INT_ARGB : src.getType();
        BufferedImage dest = new BufferedImage(newW, newH, type);
        Graphics2D g = dest.createGraphics();
        g.translate((newW - w) / 2, (newH - h) / 2);
        g.rotate(radians, w / 2.0, h / 2.0);

    private void validateAndProcessIdToken(String idToken, String expectedNonce, OAuthCallback callback) {
        try {
            String[] parts = idToken.split("\\.");
            if (parts.length != 3) {
                callback.onError("Invalid ID token format.");
                return;
            }

            String headerJson = new String(Base64.getUrlDecoder().decode(parts[0]), StandardCharsets.UTF_8);
            String payloadJson = new String(Base64.getUrlDecoder().decode(parts[1]), StandardCharsets.UTF_8);
            
            JSONObject header = new JSONObject(headerJson);
            JSONObject payload = new JSONObject(payloadJson);

            String kid = header.optString("kid", null);
            if (kid == null) {
                callback.onError("Missing kid in ID token.");
                return;
            }

            PublicKey pubKey = getGooglePublicKey(kid);
            if (pubKey == null) {
                callback.onError("Google sign-in could not be completed. Key not found.");
                return;
            }

            // Verify signature
            Signature sig = Signature.getInstance("SHA256withRSA");
            sig.initVerify(pubKey);
            sig.update((parts[0] + "." + parts[1]).getBytes(StandardCharsets.UTF_8));
            if (!sig.verify(Base64.getUrlDecoder().decode(parts[2]))) {
                callback.onError("ID token signature is invalid.");
                return;
            }

            // Verify claims
            String iss = payload.optString("iss", "");
            if (!iss.equals("accounts.google.com") && !iss.equals("https://accounts.google.com")) {
                callback.onError("Invalid issuer.");
                return;
            }
            
            String aud = payload.optString("aud", "");
            if (!aud.equals(CLIENT_ID)) {
                callback.onError("Invalid audience.");
                return;
            }
            
            long exp = payload.optLong("exp", 0);
            long now = System.currentTimeMillis() / 1000;
            if (exp < now) {
                callback.onError("ID token has expired.");
                return;
            }
            
            String tokenNonce = payload.optString("nonce", null);
            if (tokenNonce == null || !tokenNonce.equals(expectedNonce)) {
                callback.onError("Invalid nonce.");
                return;
            }
            
            String sub = payload.optString("sub", null);
            String email = payload.optString("email", null);
            boolean emailVerified = payload.optBoolean("email_verified", false);
            String name = payload.optString("name", "");
            String picture = payload.optString("picture", "");

            if (sub == null || email == null || !emailVerified) {
                callback.onError("Incomplete profile information or email not verified.");
                return;
            }

            System.out.println("[OAuth] ID token validation successful: true");
            System.out.println("[OAuth] Google identity verified: true");
            callback.onSuccess(sub, email, name, picture);

        } catch (Exception e) {
            e.printStackTrace();
            System.out.println("[OAuth] Exception during ID token validation: " + e.getMessage());
            callback.onError("Failed to validate ID token.");
        }
    }

    private PublicKey getGooglePublicKey(String kid) throws Exception {
        if (cachedJwks == null) {
            fetchJwks();
        }
        PublicKey key = findKey(kid);
        if (key == null) {
            // refresh once
            fetchJwks();
            key = findKey(kid);
        }
        return key;
    }

    private PublicKey findKey(String kid) throws Exception {
        if (cachedJwks == null) return null;
        for (int i = 0; i < cachedJwks.length(); i++) {
            JSONObject jwk = cachedJwks.getJSONObject(i);
            if (kid.equals(jwk.optString("kid"))) {
                String nStr = jwk.getString("n");
                String eStr = jwk.getString("e");
                BigInteger n = new BigInteger(1, Base64.getUrlDecoder().decode(nStr));
                BigInteger e = new BigInteger(1, Base64.getUrlDecoder().decode(eStr));
                RSAPublicKeySpec spec = new RSAPublicKeySpec(n, e);
                KeyFactory kf = KeyFactory.getInstance("RSA");
                return kf.generatePublic(spec);
            }
        }
        return null;
    }

    private void fetchJwks() throws Exception {
        URL url = new URL(JWK_URL);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod("GET");
        if (conn.getResponseCode() == 200) {
            InputStream is = conn.getInputStream();
            String jsonStr = new String(is.readAllBytes(), StandardCharsets.UTF_8);
            JSONObject root = new JSONObject(jsonStr);
            cachedJwks = root.getJSONArray("keys");
        }
    }

    private String generateRandomString(SecureRandom sr, int length) {
        byte[] bytes = new byte[length];
        sr.nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    private String generateCodeChallenge(String verifier) throws Exception {
        MessageDigest md = MessageDigest.getInstance("SHA-256");
        byte[] digest = md.digest(verifier.getBytes(StandardCharsets.US_ASCII));
        return Base64.getUrlEncoder().withoutPadding().encodeToString(digest);
    }

    private Map<String, String> parseQuery(String query) {
        Map<String, String> params = new HashMap<>();
        if (query == null) return params;
        String[] pairs = query.split("&");
        for (String pair : pairs) {
            String[] idx = pair.split("=");
            try {
                if (idx.length > 1) {
                    params.put(URLDecoder.decode(idx[0], "UTF-8"), URLDecoder.decode(idx[1], "UTF-8"));
                } else if (idx.length == 1) {
                    params.put(URLDecoder.decode(idx[0], "UTF-8"), "");
                }
            } catch (Exception e) {}
        }
        return params;
    }

    private JLabel createReqLabel(String text) {
        JLabel lbl = new JLabel(text);
        lbl.setFont(new Font("Segoe UI", Font.PLAIN, 11));
        lbl.setForeground(invalidColor);
        return lbl;
    }

    public void updateStrength(String password) {
        isLengthValid = password.length() >= 8;
        isUpperValid = password.matches(".*[A-Z].*");
        isLowerValid = password.matches(".*[a-z].*");
        isNumberValid = password.matches(".*[0-9].*");

        updateLabel(lblLength, isLengthValid, "Min 8 characters");
        updateLabel(lblUpper, isUpperValid, "1 Uppercase");
        updateLabel(lblLower, isLowerValid, "1 Lowercase");
        updateLabel(lblNumber, isNumberValid, "1 Number");
    }

    private void updateLabel(JLabel label, boolean valid, String text) {
        if (valid) {
            label.setText("✓ " + text);
            label.setForeground(validColor);
        } else {
            label.setText("✕ " + text);
            label.setForeground(invalidColor);
        }
    }

    public boolean isPasswordValid() {
        return isLengthValid && isUpperValid && isLowerValid && isNumberValid;
    }
}

    private void applyFilters() {

        String status = cbStatusFilter != null ? (String) cbStatusFilter.getSelectedItem() : "All Statuses";
        boolean evOnly = chkEvFilter != null && chkEvFilter.isSelected();

        gridPanel.removeAll();
        for (ParkingSlot slot : slots) {
            if (!query.isEmpty() && !slot.getSlotName().toLowerCase().contains(query)) {
                continue;
            }
            if (status != null && !"All Statuses".equals(status) && !status.equalsIgnoreCase(slot.getStatus())) {
                continue;
            }
            if (evOnly && !slot.isEv()) {
                continue;
            }
            gridPanel.add(createSlotCard(slot));
        }
        gridPanel.revalidate();
        gridPanel.repaint();
    }

            return;
        }

        Vehicle selectedVeh = myVehicles.get(0);
        if (myVehicles.size() > 1) {
            String[] vStrings = myVehicles.stream().map(v -> v.getVehicleNumber() + " (" + v.getVehicleType() + ")").toArray(String[]::new);
            String chosen = (String) JOptionPane.showInputDialog(this, "Select your vehicle:", "Vehicle Selection",

            for (Vehicle v : myVehicles) {
                if (chosen.startsWith(v.getVehicleNumber())) {
                    selectedVeh = v;
                    break;
                }
            }
        }

        BookingDialog dialog = new BookingDialog(parentFrame, currentUser, selectedVeh, slot);
        dialog.setVisible(true);
        refreshSlots();
    }

    private void refreshActivity() {
        activityPanel.removeAll();
        
        String sql = "SELECT u.name, v.vehicle_number, s.slot_name, p.status, p.entry_time, p.exit_time " +
                     "FROM parking_entries_exits p " +
                     "JOIN vehicles v ON p.vehicle_id = v.id " +
                     "JOIN users u ON p.user_id = u.id " +
                     "JOIN parking_slots s ON p.slot_id = s.id " +
                     "ORDER BY p.id DESC LIMIT 5";
                     
        try (Connection conn = DatabaseConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql);
             ResultSet rs = pstmt.executeQuery()) {
             
            SimpleDateFormat sdf = new SimpleDateFormat("hh:mm a");
            
            while (rs.next()) {
                String status = rs.getString("status");
                boolean isCompleted = "COMPLETED".equals(status);
                String timeStr = "";
                
                if (isCompleted && rs.getTimestamp("exit_time") != null) {
                    timeStr = sdf.format(rs.getTimestamp("exit_time"));
                } else if (rs.getTimestamp("entry_time") != null) {
                    timeStr = sdf.format(rs.getTimestamp("entry_time"));
                }
                
                String vNum = rs.getString("vehicle_number");
                String slotName = rs.getString("slot_name");
                
                String text = "✓ " + timeStr + "  " + vNum + "  → " + slotName + "  " + (isCompleted ? "EXIT" : "ENTRY");
                
                JLabel lbl = new JLabel(text);
                lbl.setFont(new Font("Consolas", Font.PLAIN, 14));
                lbl.setForeground(isCompleted ? new Color(239, 68, 68) : new Color(16, 185, 129));
                lbl.setBorder(new EmptyBorder(5, 5, 5, 5));
                
                activityPanel.add(lbl);
            }
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
        
        if (activityPanel.getComponentCount() == 0) {
            activityPanel.add(new JLabel("No recent activity."));
        }
        
        activityPanel.revalidate();
        activityPanel.repaint();
    }

    private long getDailyCount(String status) {
        // We look for either ACTIVE sessions starting today, or COMPLETED sessions matching
        String sql = "SELECT COUNT(*) FROM parking_entries_exits WHERE (status = 'ACTIVE' OR status = 'COMPLETED') AND DATE(entry_time) = CURDATE()";
        if (status.equals("EXIT")) {
            sql = "SELECT COUNT(*) FROM parking_entries_exits WHERE status = 'COMPLETED' AND DATE(exit_time) = CURDATE()";
        }
        
        try (Connection conn = DatabaseConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            ResultSet rs = pstmt.executeQuery();
            if (rs.next()) return rs.getLong(1);
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return 0;

    private String hashPassword(String plain) {
        try {
            java.security.MessageDigest md = java.security.MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(plain.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder hexString = new StringBuilder(2 * hash.length);
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) hexString.append('0');
                hexString.append(hex);
            }
            return hexString.toString();
        } catch (Exception ex) {
            throw new RuntimeException(ex);
        }
    }

    private JButton createNavButton(String text, String actionCommand, String emojiIcon) {
        boolean isActive = actionCommand.equals(activeNavCommand);

        JButton btn = new JButton("   " + text);
        btn.setIcon(new Icon() {
            @Override
            public void paintIcon(Component c, Graphics g, int x, int y) {
                g.setFont(new Font("Segoe UI Emoji", Font.PLAIN, 16));
                g.drawString(emojiIcon, x, y + 14);
            }
            @Override public int getIconWidth() { return 20; }
            @Override public int getIconHeight() { return 20; }
        });

        btn.setMaximumSize(new Dimension(216, 42));
        btn.setPreferredSize(new Dimension(216, 42));
        btn.setAlignmentX(Component.LEFT_ALIGNMENT);

        if (isActive) {
            btn.setBackground(new Color(0, 102, 255)); // Bright Blue active #0066FF
            btn.setForeground(Color.WHITE);
        } else {
            btn.setBackground(new Color(11, 25, 44));
            btn.setForeground(new Color(203, 213, 225));
        }

        btn.setFocusPainted(false);
        btn.setBorderPainted(false);
        btn.setFont(new Font("Segoe UI", Font.BOLD, 14));
        btn.setHorizontalAlignment(SwingConstants.LEFT);
        btn.setActionCommand(actionCommand);
        btn.setCursor(new Cursor(Cursor.HAND_CURSOR));
        btn.setBorder(new EmptyBorder(8, 16, 8, 16));

        btn.addMouseListener(new MouseAdapter() {
            public void mouseEntered(MouseEvent evt) {
                if (!actionCommand.equals(activeNavCommand) && !actionCommand.equals("LOGOUT")) {
                    btn.setBackground(new Color(30, 41, 59));
                    btn.setForeground(Color.WHITE);
                }
            }
            public void mouseExited(MouseEvent evt) {
                if (!actionCommand.equals(activeNavCommand) && !actionCommand.equals("LOGOUT")) {
                    btn.setBackground(new Color(11, 25, 44));
                    btn.setForeground(new Color(203, 213, 225));
                }
            }
        });

        btn.addActionListener(e -> navigateTo(e.getActionCommand()));
        return btn;
    }

    public void navigateTo(String command) {
        if ("LOGOUT".equals(command)) {
            if (securityPanel != null) securityPanel.cleanup();
            this.dispose();
            new LoginFrame().setVisible(true);
            return;
        }

        if ("QR_PASS".equals(command)) {
            List<Vehicle> myVehicles = vehicleDAO.getVehiclesByUserId(currentUser.getId());
            if (!myVehicles.isEmpty()) {
                new VehicleQRDialog(this, myVehicles.get(0), currentUser.getName()).setVisible(true);
            } else {
                JOptionPane.showMessageDialog(this, "No vehicles registered. Register a vehicle first.", "No Vehicles", JOptionPane.INFORMATION_MESSAGE);
            }
            return;
        }

        this.activeNavCommand = command;
        rebuildSidebarNav();

        if ("LIVE_MAP".equals(command)) {
            mapPanel.refreshSlots();
        } else if ("WELCOME".equals(command)) {
            contentPanel.add(new DashboardHomePanel(currentUser, this::navigateTo), "WELCOME_REFRESHED");
            cardLayout.show(contentPanel, "WELCOME_REFRESHED");
            return;
        } else if ("VEHICLES".equals(command)) {
            if (vehiclePanel != null) vehiclePanel.refreshData();
        } else if ("PROFILE".equals(command)) {
            contentPanel.add(createProfilePanel(), "PROFILE_REFRESHED");
            cardLayout.show(contentPanel, "PROFILE_REFRESHED");
            return;
        }

        cardLayout.show(contentPanel, command);
    }
